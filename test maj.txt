function onOpen() {
  SpreadsheetApp.getUi()
    .createMenu("🔄 Sync Inventaire")
    .addItem("▶️ Lancer la synchronisation", "lancerSync")
    .addItem("👁️ Aperçu sans modification", "lancerApercu")

    .addItem("⚙️ Aide", "afficherAide")

// ============================================================
//  POINT D'ENTRÉE PRINCIPAL

function lancerSync() {
  _executer(false);
}

function lancerApercu() {
  _executer(true);
}

// ============================================================
//  MENU D'ACCUEIL — Saisie de l'extract du lundi
// ============================================================
function _demanderExtract() {
  const ui = SpreadsheetApp.getUi();

  // Étape 1 : expliquer ce qu'on attend
  const intro = ui.alert(
    "📂 Nouvel extract hebdomadaire",
    "Bienvenue dans la synchronisation Discovery → Inventaire.\n\n" +
    "Vous allez indiquer où se trouve le fichier extract reçu ce lundi.\n\n" +
    "Deux options possibles :\n" +
    "  • Coller l'URL complète du fichier Google Sheets\n" +
    "  • Coller uniquement l'ID du fichier\n\n" +
    "Cliquez sur OK pour continuer.",
    ui.ButtonSet.OK_CANCEL
  );
  if (intro === ui.Button.CANCEL) return null;

  // Étape 2 : saisie de l'URL ou de l'ID
  const reponse = ui.prompt(
    "📋 Fichier extract du lundi",
    "Collez ici l'URL ou l'ID du fichier extract Google Sheets :\n\n" +
    "Exemple URL : https://docs.google.com/spreadsheets/d/1aBcD.../edit\n" +
    "Exemple ID  : 1aBcDeFgHiJkLmNoPqRsTuVwXyZ",
    ui.ButtonSet.OK_CANCEL
  );

  if (reponse.getSelectedButton() === ui.Button.CANCEL) return null;

  let saisie = reponse.getResponseText().trim();
  if (!saisie) {
    ui.alert("❌ Saisie vide", "Aucun fichier renseigné. Opération annulée.", ui.ButtonSet.OK);
    return null;
  }

  // Extraire l'ID depuis une URL si nécessaire
  const match = saisie.match(/\/spreadsheets\/d\/([a-zA-Z0-9_-]+)/);
  const extractId = match ? match[1] : saisie;

  // Étape 3 : demander le nom de l'onglet (avec valeur par défaut)
  const ongletReponse = ui.prompt(
    "📑 Nom de l'onglet dans l'extract",
    `Quel est le nom de l'onglet à utiliser dans ce fichier ?\n\n(Appuyez sur OK pour garder la valeur par défaut : "${CONFIG.EXTRACT_TAB}")`,
    ui.ButtonSet.OK_CANCEL
  );

  if (ongletReponse.getSelectedButton() === ui.Button.CANCEL) return null;

  const ongletSaisi = ongletReponse.getResponseText().trim();
  const extractTab  = ongletSaisi || CONFIG.EXTRACT_TAB;

  // Étape 4 : vérifier que le fichier est accessible
  try {
    const testSS    = SpreadsheetApp.openById(extractId);
    const testSheet = testSS.getSheetByName(extractTab);
    if (!testSheet) {
      ui.alert(
        "❌ Onglet introuvable",
        `Le fichier est accessible mais l'onglet "${extractTab}" n'existe pas.\n\nOnglets disponibles : ${testSS.getSheets().map(s => s.getName()).join(", ")}`,
        ui.ButtonSet.OK
      );
      return null;
    }
    const nbLignes = testSheet.getLastRow() - 1;

    // Étape 5 : confirmation récap
    const confirm = ui.alert(
      "✅ Fichier trouvé — Confirmer ?",
      `Fichier : ${testSS.getName()}\n` +
      `Onglet  : ${extractTab}\n` +
      `Lignes  : ${nbLignes} enregistrement(s) détecté(s)\n\n` +
      `Lancer la synchronisation avec ce fichier ?`,
      ui.ButtonSet.YES_NO
    );

    if (confirm !== ui.Button.YES) return null;

    return { extractId, extractTab };

  } catch (e) {
    ui.alert(
      "❌ Fichier inaccessible",
      `Impossible d'ouvrir le fichier.\n\nVérifiez :\n• Que l'URL ou l'ID est correct\n• Que votre compte a accès à ce fichier\n\nErreur : ${e.message}`,
      ui.ButtonSet.OK
    );
    return null;
  }
}

function _executer(apercuSeulement) {
  const ui = SpreadsheetApp.getUi();
  const label = apercuSeulement ? "APERÇU" : "SYNCHRONISATION";

  try {
    Logger.log(`=== Début ${label} ===`);

    // 0. Menu d'accueil — saisie de l'extract
    const extractInfo = _demanderExtract();
    if (!extractInfo) {
      Logger.log("Opération annulée par l'utilisateur.");
      return;
    }
    const { extractId, extractTab } = extractInfo;

    // 1. Charger les données
    // Le script tourne dans l'inventaire -> on utilise getActiveSpreadsheet()
    // plutot que openById() qui cause "Illegal spreadsheet" sur le fichier courant
    const invSS    = SpreadsheetApp.getActiveSpreadsheet();
    const extSS    = SpreadsheetApp.openById(extractId);
    const invSheet = invSS.getSheetByName(CONFIG.INVENTAIRE_TAB);
    const extSheet = extSS.getSheetByName(extractTab);

    if (!invSheet) throw new Error(`Onglet inventaire "${CONFIG.INVENTAIRE_TAB}" introuvable.`);
    if (!extSheet) throw new Error(`Onglet extract "${CONFIG.EXTRACT_TAB}" introuvable.`);

    const invData  = invSheet.getDataRange().getValues();
    const extData  = extSheet.getDataRange().getValues();

    // 2. Indexer l'inventaire par identifiant ET par nom de machine
    const invIndexParId  = {};  // INV-XXXXX -> index ligne
    const invIndexParNom = {};  // nom machine (lowercase) -> index ligne

    for (let i = 1; i < invData.length; i++) {
      const id  = String(invData[i][CONFIG.INVENTAIRE_KEY_COL - 1]).trim();
      const nom = String(invData[i][CONFIG.INVENTAIRE_NOM_COL - 1]).trim().toLowerCase();
      if (id)  invIndexParId[id]   = i;
      if (nom) invIndexParNom[nom] = i;
    }

    // 3. Parcourir l'extract et identifier les ecarts
    const actions        = [];
    const nonTrouves     = []; // Cas 5 : absent inventaire, présent discovery

    for (let e = 1; e < extData.length; e++) {
      const row = extData[e];
      const key = String(row[CONFIG.EXTRACT_KEY_COL - 1]).trim();
      if (!key) continue;

      // ── Lecture du CAS de réconciliation ──
      const cas = parseInt(String(row[CONFIG.EXTRACT_CAS_COL - 1]).trim(), 10);

      // Cas 5 : absent de l'inventaire, présent dans discovery
      // → on collecte le nom de la machine pour la popup, on ne touche rien
      if (cas === 5) {
        const nomMachine = String(row[CONFIG.EXTRACT_NOM_COL - 1]).trim() || key;
        nonTrouves.push({ key, nom: nomMachine });
        Logger.log(`CAS 5 : "${key}" (${nomMachine}) absent de l'inventaire — à créer manuellement`);
        continue;
      }

      // Cas 3 & 4 : présent dans l'inventaire, absent de discovery
      // → aucune mise à jour, les données discovery ne sont pas fiables
      if (cas === 3 || cas === 4) {
        Logger.log(`CAS ${cas} : "${key}" absent de discovery — aucune modification`);
        continue;
      }

      // Cas 1 & 2 : présent des deux côtés → on traite les écarts normalement
      let invRowIdx = undefined;
      let modeMatch = "";

      if (/^INV-/i.test(key) && invIndexParId[key] !== undefined) {
        invRowIdx = invIndexParId[key];
        modeMatch = "id";
      } else {
        const nomExtract = String(row[CONFIG.EXTRACT_NOM_COL - 1]).trim().toLowerCase();
        if (nomExtract && invIndexParNom[nomExtract] !== undefined) {
          invRowIdx = invIndexParNom[nomExtract];
          modeMatch = "nom";
          Logger.log(`INFO : Cle "${key}" non trouvee par ID, correspondance par nom "${nomExtract}"`);
        }
      }

      if (invRowIdx === undefined) {
        Logger.log(`SKIP : "${key}" (cas ${cas}) — introuvable dans l'inventaire ni par ID ni par nom`);
        continue;
      }

      // Pour chaque champ mappé, vérifier la colonne match
      for (const [champ, matchCol] of Object.entries(CONFIG.MATCH_COLS)) {
        const matchVal = String(row[matchCol - 1] ?? "").trim();

        // ✅ Coche verte  → inventaire = discovery, rien à faire
        // –  Tiret        → pas de donnée discovery, rien à faire
        // ❌ Croix rouge  → différence détectée, on met à jour
        const estOk    = matchVal === "✅" || matchVal === "☑" || matchVal === "✔" || matchVal === "✓";
        const estTiret = matchVal === "–"  || matchVal === "-" || matchVal === "—" || matchVal === "";
        const estEcart = matchVal === "❌" || matchVal === "✗" || matchVal === "✘" || matchVal === "×";

        if (estOk || estTiret) continue;
        if (!estEcart) {
          Logger.log(`⚠️ Valeur de match inconnue pour "${champ}" (clé ${key}) : "${matchVal}" — ignoré`);
          continue;
        }

        let discoveryVal = String(row[CONFIG.MAPPING[champ].col - 1] ?? "").trim();

        // Sécurité anti-écrasement : si la valeur discovery est vide ou N/A,
        // on ne touche pas une donnée cohérente de l'inventaire
        if (!discoveryVal) continue;
        const discoveryNorm = discoveryVal.toUpperCase();
        if (discoveryNorm === "N/A" || discoveryNorm === "NA" || discoveryNorm === "-") {
          Logger.log(`SKIP N/A : ${key} / ${champ} — valeur discovery "${discoveryVal}" ignorée`);
          continue;
        }

        // Regle 1 : "DISCOVERED" (toutes variantes de casse) est remplace par 1
        if (discoveryNorm === "DISCOVERED") discoveryVal = "1";

        const invColIdx   = CONFIG.INVENTAIRE_COLS[champ] - 1;
        const invActuelle = String(invData[invRowIdx][invColIdx] ?? "").trim();

        // Regle 2 : comparaison insensible a la casse
        if (discoveryVal.toLowerCase() === invActuelle.toLowerCase()) continue;

        actions.push({
          key,
          cas,
          champ,
          ancienne:    invActuelle,
          nouvelle:    discoveryVal,
          invRowSheet: invRowIdx + 1,
          invColSheet: CONFIG.INVENTAIRE_COLS[champ],
          modeMatch,
        });
      }
    }

    if (actions.length === 0) {
      ui.alert("✅ Déjà à jour", "Aucune différence détectée entre l'extract et l'inventaire.", ui.ButtonSet.OK);
      return;
    }

    // 4. Alerte nouvelles machines (Cas 5) — avant tout le reste
    if (nonTrouves.length > 0) {
      const listeMachines = nonTrouves
        .map(m => `  • ${m.nom}${m.nom !== m.key ? "  (ID : " + m.key + ")" : ""}`)
        .join("\n");
      ui.alert(
        "🆕 Cas 5 — Nouvelles machines à créer dans l'inventaire",
        `${nonTrouves.length} machine(s) présente(s) dans Discovery n'ont pas de correspondance dans l'inventaire.\n\n` +
        `Ces machines doivent être ajoutées manuellement :\n\n` +
        `${listeMachines}\n\n` +
        `ℹ️ Ces entrées sont également listées dans le rapport final.\n` +
        `La synchronisation des autres machines continue.`,
        ui.ButtonSet.OK
      );
    }

    // 5. Afficher l'aperçu
    const resume = _genererResume(actions);

    if (apercuSeulement) {
      ui.alert("👁️ Aperçu des mises à jour", resume + "\n\nAucune modification effectuée.", ui.ButtonSet.OK);
      _ecrireRapport(invSS, actions, false, nonTrouves);
      return;
    }

    // 6. Confirmation
    const conf = ui.alert(
      "⚠️ Confirmer la synchronisation",
      resume + "\n\nCes modifications seront appliquées à l'inventaire.\nContinuer ?",
      ui.ButtonSet.YES_NO
    );
    if (conf !== ui.Button.YES) {
      ui.alert("Annulé", "Aucune modification effectuée.", ui.ButtonSet.OK);
      return;
    }

    // 6. Appliquer les modifications avec notification en temps réel
    _appliquerModifications(invSheet, actions);

    // 7. Générer le rapport
    _ecrireRapport(invSS, actions, true, nonTrouves);

    // 8. Envoyer la carte Google Chat
    _envoyerGoogleChatCard(actions);

    ui.alert(
      "✅ Synchronisation terminée",
      `${actions.length} champ(s) mis à jour dans l'inventaire.\n\nUn rapport complet a été généré dans l'onglet "${CONFIG.RAPPORT_TAB}".\nUne notification a été envoyée sur Google Chat.`,
      ui.ButtonSet.OK
    );

  } catch (e) {
    Logger.log("ERREUR : " + e.message);
    ui.alert("❌ Erreur", e.message, ui.ButtonSet.OK);
  }
}

// ============================================================
//  APPLIQUER LES MODIFICATIONS (avec toast en temps réel)
// ============================================================
function _appliquerModifications(invSheet, actions) {
  const ss      = invSheet.getParent();
  const dateMAJ = new Date().toLocaleString("fr-FR");

  ss.toast("Lecture de l'inventaire...", "Synchronisation en cours", 10);

  // ── 1. Lire toute la plage inventaire EN UNE FOIS en memoire ──────────────
  const lastRow  = invSheet.getLastRow();
  const lastCol  = invSheet.getLastColumn();
  const plage    = invSheet.getRange(1, 1, lastRow, lastCol);
  const valeurs  = plage.getValues();       // tableau 2D complet en memoire
  const couleurs = plage.getBackgrounds();  // couleurs actuelles
  const notes    = plage.getNotes();        // notes actuelles

  // ── 2. Lire les validations des colonnes concernees (une par colonne unique) ─
  ss.toast("Lecture des validations...", "Synchronisation en cours", 10);
  const colsUniques = [...new Set(actions.map(a => a.invColSheet))];
  const validationParCol = {};
  for (const col of colsUniques) {
    try {
      // On lit la validation sur la ligne 2 (premiere ligne de donnees)
      validationParCol[col] = invSheet.getRange(2, col).getDataValidation();
    } catch(e) {
      validationParCol[col] = null;
    }
  }

  // ── 3. Calculer les valeurs finales et modifier le tableau en memoire ──────
  ss.toast("Calcul des modifications...", "Synchronisation en cours", 5);

  for (const action of actions) {
    const validation = validationParCol[action.invColSheet];
    const rowIdx     = action.invRowSheet - 1; // 0-index pour le tableau
    const colIdx     = action.invColSheet - 1;

    try {
      let valeurAEcrire = action.nouvelle;
      action.horsListe  = false;

      if (validation) {
        const criteria = validation.getCriteriaType();
        const isDdl    = criteria === SpreadsheetApp.DataValidationCriteria.VALUE_IN_LIST
                      || criteria === SpreadsheetApp.DataValidationCriteria.VALUE_IN_RANGE;

        if (isDdl && criteria === SpreadsheetApp.DataValidationCriteria.VALUE_IN_LIST) {
          const valeursAutorisees = validation.getCriteriaValues()[0] || [];
          const valeurNorm        = String(action.nouvelle).trim();
          const valeurDansListe   = valeursAutorisees.find(
            v => String(v).trim().toLowerCase() === valeurNorm.toLowerCase()
          );

          if (valeurDansListe !== undefined) {
            valeurAEcrire = String(valeurDansListe).trim();
            if (valeurAEcrire !== action.nouvelle) {
              action.avertissement = `Casse normalisee : "${action.nouvelle}" -> "${valeurAEcrire}"`;
              action.nouvelle      = valeurAEcrire;
            }
          } else {
            // Hors liste : on ecrit quand meme
            action.horsListe     = true;
            action.avertissement = `Valeur "${action.nouvelle}" hors liste deroulante - ecrite quand meme`;
            Logger.log(`HORS LISTE : ${action.key} / ${action.champ} = "${action.nouvelle}"`);
          }
        }
      }

      // Modifier le tableau en memoire
      valeurs[rowIdx][colIdx]  = valeurAEcrire;
      action.valeurFinale      = valeurAEcrire;
      action.statut            = action.horsListe ? "OK_HORS_LISTE" : "OK";

      // Couleur dans le tableau memoire
      couleurs[rowIdx][colIdx] = action.horsListe ? "#FFE0B2" : "#FFF59D";

      // Note dans le tableau memoire
      let note = `Mis a jour par Discovery le ${dateMAJ}
Ancienne valeur : "${action.ancienne}"`;
      if (action.avertissement) note += `
${action.avertissement}`;
      notes[rowIdx][colIdx] = note;

    } catch(e) {
      action.statut       = "ECHEC";
      action.erreur       = e.message;
      action.valeurFinale = null;
      couleurs[rowIdx][colIdx] = "#FFCDD2";
      notes[rowIdx][colIdx]    = `Echec le ${dateMAJ} : ${e.message}`;
      Logger.log(`ECHEC : ${action.key} / ${action.champ} - ${e.message}`);
    }
  }

  // ── 4. Ecrire toute la plage EN UN SEUL APPEL API ─────────────────────────
  ss.toast("Ecriture dans l'inventaire...", "Synchronisation en cours", 15);

  // Pour les cellules hors liste : desactiver la validation sur ces colonnes le temps d'ecrire
  const colsHorsListe = [...new Set(
    actions.filter(a => a.horsListe).map(a => a.invColSheet)
  )];
  for (const col of colsHorsListe) {
    invSheet.getRange(2, col, lastRow - 1, 1).clearDataValidations();
  }

  // Ecriture des valeurs, couleurs et notes en 3 appels globaux
  plage.setValues(valeurs);
  plage.setBackgrounds(couleurs);
  plage.setNotes(notes);

  // Remettre les validations sur les colonnes hors liste
  for (const col of colsHorsListe) {
    const validation = validationParCol[col];
    if (validation) invSheet.getRange(2, col, lastRow - 1, 1).setDataValidation(validation);
  }

  // ── 5. Bilan ──────────────────────────────────────────────────────────────
  const ok        = actions.filter(a => a.statut === "OK").length;
  const horsListe = actions.filter(a => a.statut === "OK_HORS_LISTE").length;
  const echecs    = actions.filter(a => a.statut === "ECHEC").length;

  let bilanMsg = `${ok} mise(s) a jour OK`;
  if (horsListe > 0) bilanMsg += ` | ${horsListe} hors liste deroulante`;
  if (echecs    > 0) bilanMsg += ` | ${echecs} echec(s)`;

  ss.toast(bilanMsg, "Synchronisation terminee", 8);
}


// ============================================================
//  RÉSUMÉ TEXTE (pour boîte de dialogue)
// ============================================================
function _genererResume(actions) {
  // Grouper par clé
  const parCle = {};
  for (const a of actions) {
    if (!parCle[a.key]) parCle[a.key] = [];
    parCle[a.key].push(a);
  }

  const nbCles = Object.keys(parCle).length;
  let msg = `📊 ${actions.length} différence(s) trouvée(s) sur ${nbCles} serveur(s) :\n`;
  msg += "─".repeat(45) + "\n";

  let shown = 0;
  for (const [key, items] of Object.entries(parCle)) {
    if (shown >= 10) { msg += `\n... et ${nbCles - shown} autre(s) serveur(s)`; break; }
    msg += `\n🖥️ ${key} (${items.length} champ(s))\n`;
    for (const item of items.slice(0, 4)) {
      msg += `   • ${item.champ}: "${item.ancienne}" → "${item.nouvelle}"\n`;
    }
    if (items.length > 4) msg += `   ... +${items.length - 4} autre(s)\n`;
    shown++;
  }

  return msg;
}

// ============================================================
//  RAPPORT COMPLET
// ============================================================
function _ecrireRapport(invSS, actions, applique, nonTrouves) {
  let sheet = invSS.getSheetByName(CONFIG.RAPPORT_TAB);
  if (sheet) { sheet.clearContents(); sheet.clearFormats(); }
  else sheet = invSS.insertSheet(CONFIG.RAPPORT_TAB);

  const now    = new Date();
  const statut = applique ? "✅ APPLIQUÉ" : "👁️ APERÇU (non appliqué)";
  const couleurStatut = applique ? "#34A853" : "#FBBC05";

  // ── Titre ──
  sheet.getRange("A1:G1").merge()
    .setValue(`Rapport de synchronisation Discovery → Inventaire`)
    .setFontSize(14).setFontWeight("bold")
    .setBackground("#1a73e8").setFontColor("white")
    .setHorizontalAlignment("center");

  sheet.getRange("A2:G2").merge()
    .setValue(`${now.toLocaleString("fr-FR")}   |   Statut : ${statut}`)
    .setBackground(couleurStatut).setFontColor("white")
    .setHorizontalAlignment("center").setFontWeight("bold");

  // ── Résumé chiffré ──
  const parCle   = {};
  const parChamp = {};
  for (const a of actions) {
    parCle[a.key]     = (parCle[a.key]     || 0) + 1;
    parChamp[a.champ] = (parChamp[a.champ] || 0) + 1;
  }

  sheet.getRange("A4:B4").setValues([["Serveurs impactés", Object.keys(parCle).length]])
    .setFontWeight("bold");
  sheet.getRange("A5:B5").setValues([["Champs mis à jour", actions.length]]);
  sheet.getRange("A6:B6").setValues([["Date de traitement", now.toLocaleDateString("fr-FR")]]);

  // ── En-têtes du détail ──
  const headers = ["Identifiant", "Champ modifié", "Ancienne valeur (Inventaire)", "Nouvelle valeur (Discovery)", "Correspondance", "Statut"];
  const hRow = 9;
  sheet.getRange(hRow, 1, 1, headers.length).setValues([headers])
    .setFontWeight("bold").setBackground("#E8F0FE")
    .setBorder(true, true, true, true, true, true);

  // ── Données ──
  const rows = actions.map(a => [
    a.key,
    a.champ,
    a.ancienne,
    a.nouvelle,
    a.modeMatch === "nom" ? "Par nom (nouvelle machine)" : "Par ID",
    (() => {
          if (!applique) return "En attente";
          if (a.statut === "ECHEC")         return "ECHEC - " + (a.erreur || "");
          if (a.statut === "OK_HORS_LISTE") return "Hors liste DDL";
          return "OK";
        })(),
  ]);

  if (rows.length > 0) {
    sheet.getRange(hRow + 1, 1, rows.length, headers.length).setValues(rows);

    // Zebra stripes
    for (let i = 0; i < rows.length; i++) {
      const bg = i % 2 === 0 ? "#FFFFFF" : "#F8F9FA";
      sheet.getRange(hRow + 1 + i, 1, 1, headers.length).setBackground(bg);
    }
  }

  // ── Répartition par champ ──
  let startCol = 9;
  sheet.getRange(4, startCol).setValue("Champ").setFontWeight("bold").setBackground("#E8F0FE");
  sheet.getRange(4, startCol + 1).setValue("Nb écarts").setFontWeight("bold").setBackground("#E8F0FE");
  let r = 5;
  for (const [champ, nb] of Object.entries(parChamp).sort((a, b) => b[1] - a[1])) {
    sheet.getRange(r, startCol).setValue(champ);
    sheet.getRange(r, startCol + 1).setValue(nb);
    r++;
  }

  // Section machines Cas 5 (non trouvées dans l'inventaire)
  if (nonTrouves && nonTrouves.length > 0) {
    let lastRow = sheet.getLastRow() + 2;
    sheet.getRange(lastRow, 1, 1, 3)
      .setValues([["🆕 Cas 5 — Machines absentes de l'inventaire (à créer manuellement)", "", ""]])
      .setFontWeight("bold").setBackground("#FFCDD2");
    sheet.getRange(lastRow, 1, 1, 3).merge();
    lastRow++;
    // En-têtes
    sheet.getRange(lastRow, 1, 1, 2).setValues([["Nom machine", "Clé Discovery"]])
      .setFontWeight("bold").setBackground("#FFEBEE");
    lastRow++;
    nonTrouves.forEach(m => {
      sheet.getRange(lastRow, 1).setValue(m.nom);
      sheet.getRange(lastRow, 2).setValue(m.key);
      lastRow++;
    });
  }

  sheet.autoResizeColumns(1, headers.length + 2);

  Logger.log(`Rapport écrit : ${rows.length} ligne(s) dans l'onglet "${CONFIG.RAPPORT_TAB}"`);
}

// ============================================================
//  NOTIFICATION GOOGLE CHAT — CARD V2
// ============================================================
function _envoyerGoogleChatCard(actions) {
  const now      = new Date();
  const parCle   = {};
  const parChamp = {};

  for (const a of actions) {
    if (!parCle[a.key]) parCle[a.key] = [];
    parCle[a.key].push(a);
    parChamp[a.champ] = (parChamp[a.champ] || 0) + 1;
  }

  const nbServeurs = Object.keys(parCle).length;

  // ── Widgets de répartition par champ ──
  const champWidgets = Object.entries(parChamp)
    .sort((a, b) => b[1] - a[1])
    .map(([champ, nb]) => ({
      decoratedText: {
        topLabel: champ,
        text: `${nb} mise(s) à jour`,
        startIcon: { knownIcon: "BOOKMARK" },
      }
    }));

  // ── Widgets de détail des modifications (max 15 pour ne pas dépasser la taille) ──
  const detailWidgets = actions.slice(0, 15).map(a => ({
    decoratedText: {
      topLabel: `${a.key} — ${a.champ}`,
      text: `<font color="#c62828">${a.ancienne || "(vide)"}</font>  →  <font color="#2e7d32">${a.nouvelle}</font>`,
    }
  }));

  if (actions.length > 15) {
    detailWidgets.push({
      textParagraph: {
        text: `<i>... et ${actions.length - 15} autre(s) modification(s). Consultez le rapport dans l'onglet "${CONFIG.RAPPORT_TAB}".</i>`
      }
    });
  }

  // ── Construction de la Card v2 ──
  const card = {
    cardsV2: [{
      cardId: "sync-rapport",
      card: {
        header: {
          title: "🔄 Sync Discovery → Inventaire",
          subtitle: now.toLocaleString("fr-FR"),
          imageUrl: "https://fonts.gstatic.com/s/i/googlematerialicons/sync/v6/googlematerialicons-sync-48dp.png",
          imageType: "CIRCLE",
        },
        sections: [
          {
            header: "📊 Résumé",
            collapsible: false,
            widgets: [
              {
                columns: {
                  columnItems: [
                    {
                      horizontalSizeStyle: "FILL_AVAILABLE_SPACE",
                      widgets: [{
                        decoratedText: {
                          topLabel: "Serveurs impactés",
                          text: String(nbServeurs),
                          startIcon: { knownIcon: "COMPUTER" },
                        }
                      }]
                    },
                    {
                      horizontalSizeStyle: "FILL_AVAILABLE_SPACE",
                      widgets: [{
                        decoratedText: {
                          topLabel: "Champs mis à jour",
                          text: String(actions.length),
                          startIcon: { knownIcon: "EDIT" },
                        }
                      }]
                    }
                  ]
                }
              }
            ]
          },
          {
            header: "📈 Répartition par champ",
            collapsible: true,
            uncollapsibleWidgetsCount: 3,
            widgets: champWidgets,
          },
          {
            header: "📋 Détail des modifications",
            collapsible: true,
            uncollapsibleWidgetsCount: 3,
            widgets: detailWidgets,
          },
          {
            widgets: [{
              buttonList: {
                buttons: [{
                  text: "Ouvrir l'inventaire",
                  icon: { knownIcon: "OPEN_IN_NEW" },
                  onClick: {
                    openLink: {
                      url: SpreadsheetApp.getActiveSpreadsheet().getUrl(),
                    }
                  }
                }]
              }
            }]
          }
        ]
      }
    }]
  };

  const options = {
    method: "post",
    contentType: "application/json",
    payload: JSON.stringify(card),
    muteHttpExceptions: true,
  };

  const response = UrlFetchApp.fetch(CONFIG.GOOGLE_CHAT_WEBHOOK, options);
  const code     = response.getResponseCode();

  if (code !== 200) {
    Logger.log(`⚠️ Google Chat webhook — réponse HTTP ${code} : ${response.getContentText()}`);
  } else {
    Logger.log("Notification Google Chat envoyée avec succès.");
  }
}


// ============================================================
//  OUVRIR RAPPORT
// ============================================================
function ouvrirRapport() {
  const ui = SpreadsheetApp.getUi();
  try {
    const invSS = SpreadsheetApp.getActiveSpreadsheet();
    const r     = invSS.getSheetByName(CONFIG.RAPPORT_TAB);
    if (!r) {
      ui.alert("Aucun rapport", "Lancez d'abord une synchronisation ou un aperçu.", ui.ButtonSet.OK);
    } else {
      ui.alert("📋 Rapport disponible", `Consultez l'onglet "${CONFIG.RAPPORT_TAB}" dans l'inventaire.`, ui.ButtonSet.OK);
    }
  } catch(e) { ui.alert("Erreur", e.message, ui.ButtonSet.OK); }
}
