function diagnosticR71() {
  const ss    = SpreadsheetApp.getActiveSpreadsheet();
  const sheet = ss.getSheetByName("Inventaire serveurs");
  const cell  = sheet.getRange("R71");

  Logger.log("Valeur actuelle R71 : " + cell.getValue());

  const v = cell.getDataValidation();
  if (v) {
    const type   = v.getCriteriaType();
    const values = v.getCriteriaValues();
    Logger.log("Type validation : " + type);
    Logger.log("Valeurs autorisées : " + JSON.stringify(values));
    Logger.log("Rejeter si invalide : " + !v.getAllowInvalid());
  } else {
    Logger.log("Aucune validation sur R71");
  }

  // Test : est-ce qu'on arrive à désactiver la DDL et écrire ?
  const valeurTest = cell.getValue(); // on réécrit la même valeur
  cell.clearDataValidations();
  cell.setValue(valeurTest);
  Logger.log("Écriture test OK avec clearDataValidations() cellule par cellule");
}
