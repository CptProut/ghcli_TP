const valeurDansListe = valeursAutorisees.find(
  v => String(v).trim().toLowerCase() === String(action.nouvelle).trim().toLowerCase()
);

// Normalisation de casse dans tous les cas, même si DDL en mode avertissement
valeurAEcrire = valeurDansListe !== undefined
  ? String(valeurDansListe).trim()
  : action.nouvelle;

if (valeurDansListe !== undefined && valeurAEcrire !== action.nouvelle) {
  action.avertissement = `Casse normalisée : "${action.nouvelle}" → "${valeurAEcrire}"`;
  action.nouvelle      = valeurAEcrire;
} else if (valeurDansListe === undefined) {
  action.horsListe     = true;
  action.avertissement = `"${action.nouvelle}" hors liste — écrit quand même`;
}
