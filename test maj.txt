#!/usr/bin/env bash
#
# Import d'utilisateurs Nextcloud via occ, a lancer sur la machine Nextcloud, en root.
# Pas de mot de passe a gerer : l'utilisateur recoit un mail avec un lien pour choisir le sien.
#
# Usage : ./nc_import_occ.sh users.csv
#
# CSV en UTF-8, separateur ;, premiere ligne ignoree, colonnes dans cet ordre :
#   userid;email;displayname;groups;quota;language
# userid et email sont obligatoires. Plusieurs groupes : partenaires|finance

# A adapter : utilisateur du serveur web et chemin de occ
OCC=(sudo -u apache php /var/www/html/nextcloud/occ)

CSV="${1:?Usage : $0 users.csv}"
[ -f "$CSV" ] || { echo "Fichier introuvable : $CSV"; exit 1; }

# </dev/null : sinon occ avale la suite du CSV et la boucle s'arrete au premier compte
occ() { "${OCC[@]}" "$@" </dev/null 2>&1; }

occ user:add --help | grep -q -- '--generate-password' ||
    { echo "occ injoignable, ou version sans --generate-password : verifie OCC en haut du script"; exit 1; }

ok=0 ignores=0 ko=0

while IFS=';' read -r uid email name groups quota lang _ || [ -n "$uid" ]; do
    [ -n "$uid$email" ] || continue
    if [ -z "$uid" ] || [ -z "$email" ]; then
        echo "KO      $uid $email : userid et email obligatoires"; ko=$((ko + 1)); continue
    fi

    args=(--generate-password --email "$email")
    [ -z "$name" ] || args+=(--display-name "$name")
    IFS='|' read -r -a group_list <<< "$groups"
    for group in "${group_list[@]}"; do args+=(--group "$group"); done

    output=$(occ user:add "${args[@]}" "$uid")

    case "$output" in
        *"already exists"*)
            echo "IGNORE  $uid : existe deja"; ignores=$((ignores + 1)); continue ;;
        *"Welcome email sent"*)
            echo "OK      $uid : cree, mail envoye"; ok=$((ok + 1)) ;;
        *"was created successfully"*)
            echo "KO      $uid : cree SANS mail : $(printf '%s' "$output" | tail -n 1)"; ko=$((ko + 1)) ;;
        *)
            echo "KO      $uid : $output"; ko=$((ko + 1)); continue ;;
    esac

    if [ -n "$quota" ]; then occ user:setting "$uid" files quota "$quota" >/dev/null || echo "        $uid : quota non applique"; fi
    if [ -n "$lang" ];  then occ user:setting "$uid" core lang "$lang" >/dev/null || echo "        $uid : langue non appliquee"; fi
done < <(tail -n +2 "$CSV" | tr -d '\r')

echo
echo "Termine : $ok cree(s), $ignores ignore(s), $ko en erreur"
[ "$ko" -eq 0 ]
