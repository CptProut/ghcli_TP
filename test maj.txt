function doPost(e) {
  // Récupère l'événement envoyé par Google Chat
  var event = JSON.parse(e.postData.contents);
  var reply = {};

  // Gère l'événement selon son type
  if (event.type === 'ADDED_TO_SPACE') {
    reply = { "text": "Salut ! Merci de m'avoir ajouté." };
  } else if (event.type === 'MESSAGE') {
    var name = event.user.displayName;
    var text = event.message.text;
    reply = { "text": name + " a dit : " + text };
  }

  // Renvoie la réponse au bon format HTTP pour Google Chat
  return ContentService.createTextOutput(JSON.stringify(reply))
    .setMimeType(ContentService.MimeType.JSON);
}

// On garde tes anciennes fonctions au cas où
function onAddedToSpace(event) { return doPost({postData: {contents: JSON.stringify(event)}}); }
function onMessage(event) { return doPost({postData: {contents: JSON.stringify(event)}}); }
